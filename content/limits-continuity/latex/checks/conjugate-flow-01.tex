\begin{bgcheck}{conjugate-flow-01}{rational}{1/4}[Check your understanding]
Evaluate \(\lim_{x\to0}\frac{\sqrt{x+4}-2}{x}\).
\begin{bghint}
Use the conjugate \(\sqrt{x+4}+2\).
\end{bghint}
\begin{bgreveal}
The expression simplifies to \(1/(\sqrt{x+4}+2)\), giving \(1/4\).
\end{bgreveal}
\end{bgcheck}
